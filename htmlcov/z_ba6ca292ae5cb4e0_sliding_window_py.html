23def _leading_system_end(messages: list[ChatMessage]) -> int: 

24 """Return the end of the leading system-message block.""" 

25 for index, message in enumerate(messages): 

26 # Find the first non-leading system message or a system message produced by compaction 

27 if not message.is_from(role=ChatRole.SYSTEM) or _COMPACTION_META_KEY in message.meta: 

28 return index 

29 return len(messages)

32def _latest_user_index(messages: list[ChatMessage]) -> int | None: 

33 """ 

34 Return the latest user message not produced by compaction. 

35 

36 :param messages: The conversation to analyze, oldest to newest. 

37 """ 

38 # We loop backwards to find the latest user message 

39 for index in reversed(range(len(messages))): 

40 message = messages[index] 

41 # Find the latest user message that was not produced by a previous compaction 

42 if message.is_from(role=ChatRole.USER) and _COMPACTION_META_KEY not in message.meta: 

43 return index 

44 return None

47def _agent_step_spans(messages: list[ChatMessage], start: int) -> list[tuple[int, int]]: 

48 """ 

49 Return spans containing an assistant message and all immediately following tool results. 

50 

51 :param messages: The conversation to analyze, oldest to newest. 

52 :param start: The index to start searching for steps from. We recommend starting after the latest user message, or 

53 after the leading system messages when there is no user message. Otherwise, the returned spans may include 

54 steps that are not part of the current task. 

55 :returns: A list of spans, where each span is a tuple of (start_index, end_index) and end_index is exclusive. 

56 """ 

57 # e.g. a span of (2, 5) means messages[2:5] are part of the same step 

58 spans: list[tuple[int, int]] = [] 

59 index = start 

60 while index < len(messages): 

61 # Only assistant messages can start a step; skip any other messages. 

62 if not messages[index].is_from(role=ChatRole.ASSISTANT): 

63 index += 1 

64 continue 

65 # Find the end of the step by looking for the first message that is not a tool result. 

66 end = index + 1 

67 while end < len(messages) and messages[end].tool_call_results: 

68 end += 1 

69 # Record the span and continue searching for the next step. 

70 spans.append((index, end)) 

71 # Reset the index to the end of the current step to avoid overlapping spans. 

72 index = end 

73 return spans

76def _task_and_step_split( 

77 messages: list[ChatMessage], target_tokens: int, token_counter: TokenCounter, min_keep_steps: int 

78) -> tuple[list[ChatMessage], list[ChatMessage], list[ChatMessage], int]: 

79 """Split messages into the protected task context, removable history, and retained Agent steps.""" 

80 # Find the leading system messages that contain the Agent instructions. 

81 system_end = _leading_system_end(messages=messages) 

82 # Find the latest user message to use as the current task anchor. 

83 task_index = _latest_user_index(messages=messages) 

84 task = [messages[task_index]] if task_index is not None else [] 

85 # Find complete Agent steps after the current task, or after the system messages when there is no task anchor. 

86 steps = _agent_step_spans(messages=messages, start=(task_index + 1) if task_index is not None else system_end) 

87 

88 # Protect the Agent instructions and current task from removal. 

89 protected = [*messages[:system_end], *task] 

90 # The remaining token budget to retain recent Agent steps. 

91 available_tokens = target_tokens - token_counter.count(messages=protected) 

92 # Work backwards through the steps, keeping as many as fit in the remaining budget. 

93 kept_step_start = len(steps) 

94 while kept_step_start > 0: 

95 start, end = steps[kept_step_start - 1] 

96 step_tokens = token_counter.count(messages=messages[start:end]) 

97 # Stop at the first step that does not fit 

98 if step_tokens > available_tokens: 

99 break 

100 available_tokens -= step_tokens 

101 kept_step_start -= 1 

102 

103 # Enforce the minimum number of complete steps, even when they exceed the target token budget. 

104 kept_step_start = min(kept_step_start, max(len(steps) - min_keep_steps, 0)) 

105 kept_spans = steps[kept_step_start:] 

106 

107 # Record every protected or retained message index; equal ChatMessages can appear more than once in the list. 

108 kept_indices = {*range(system_end)} 

109 if task_index is not None: 

110 kept_indices.add(task_index) 

111 for start, end in kept_spans: 

112 kept_indices.update(range(start, end)) 

113 

114 # Everything outside the protected context and retained steps can be removed. 

115 removable = [message for index, message in enumerate(messages) if index not in kept_indices] 

116 # Flatten the retained spans back into the message list expected by the compactor. 

117 kept_steps = [message for start, end in kept_spans for message in messages[start:end]] 

118 return protected, removable, kept_steps, len(kept_spans)

121@_experimental 

122class SlidingWindowCompactor(Compactor): 

123 """ 

124 Keeps the Agent's instructions, current task, and as many complete recent steps as the target allows. 

125 

126 Leading system messages and the latest user message are protected. Recent history is retained in complete Agent 

127 steps, where a step is an assistant message together with all immediately following tool results. An 

128 `omission_note` is left in place of what was removed. 

129 

130 ```python 

131 from haystack.components.agents import Agent 

132 from haystack.components.generators.chat import OpenAIResponsesChatGenerator 

133 from haystack.hooks.compaction import ContextCompactionHook, SlidingWindowCompactor 

134 

135 hook = ContextCompactionHook( 

136 compactor=SlidingWindowCompactor(), context_window=200_000, compact_at=0.7, compact_to=0.4 

137 ) 

138 agent = Agent( 

139 chat_generator=OpenAIResponsesChatGenerator(model="gpt-5.4-nano"), 

140 tools=[web_search], 

141 hooks={"before_llm": [hook]}, 

142 ) 

143 ``` 

144 """ 

145 

146 def __init__(self, *, min_keep_steps: int = 1, omission_note: str | None = _DEFAULT_OMISSION_NOTE) -> None: 

147 """ 

148 Initialize the compactor. 

149 

150 :param min_keep_steps: The fewest complete recent Agent steps to keep even when they exceed the target. A step 

151 is an assistant message and all immediately following tool results. `0` allows all completed steps to be 

152 removed when none fit. 

153 :param omission_note: The user message left in place of what was removed, or None to remove the messages 

154 silently. Include `{num_removed}` to have the number of removed messages substituted in. 

155 :raises ValueError: If `min_keep_steps` is negative. 

156 """ 

157 if min_keep_steps < 0: 

158 raise ValueError(f"`min_keep_steps` must be at least 0, got {min_keep_steps}.") 

159 self.min_keep_steps = min_keep_steps 

160 self.omission_note = omission_note 

161 

162 def compact( 

163 self, messages: list[ChatMessage], target_tokens: int, token_counter: TokenCounter 

164 ) -> list[ChatMessage] | None: 

165 """ 

166 Drop older history while preserving the task anchor and a recent window of complete Agent steps. 

167 

168 :param messages: The conversation to compact, oldest to newest. 

169 :param target_tokens: The size the retained conversation should come in under. 

170 :param token_counter: The `TokenCounter` to measure messages with. 

171 :returns: The protected context, an omission note if configured, and the retained steps; or None when there is 

172 nothing worth removing. 

173 """ 

174 if token_counter.count(messages) <= target_tokens: 

175 return None 

176 protected, removable, kept_steps, kept_step_count = _task_and_step_split( 

177 messages=messages, 

178 target_tokens=target_tokens, 

179 token_counter=token_counter, 

180 min_keep_steps=self.min_keep_steps, 

181 ) 

182 if not removable: 

183 return None 

184 if not self.omission_note: 

185 return [*protected, *kept_steps] 

186 

187 # We prefer user over system since not all providers support multiple system messages 

188 note = ChatMessage.from_user( 

189 # `replace` rather than `format`, so a note carrying braces of its own cannot raise. 

190 self.omission_note.replace(_NUM_REMOVED_PLACEHOLDER, str(len(removable))), 

191 meta={ 

192 _COMPACTION_META_KEY: { 

193 "strategy": "sliding_window", 

194 "removed_messages": len(removable), 

195 "kept_messages": len(protected) + len(kept_steps), 

196 "kept_steps": kept_step_count, 

197 } 

198 }, 

199 ) 

200 # The note costs tokens of its own, so it is only worth leaving behind if what it stands in for is bigger. 

201 if token_counter.count([note]) >= token_counter.count(removable): 

202 return None 

203 return [*protected, note, *kept_steps]

211 return default_to_dict(self, min_keep_steps=self.min_keep_steps, omission_note=self.omission_note)
