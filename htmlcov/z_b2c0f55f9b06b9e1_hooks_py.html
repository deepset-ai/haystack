21# Meta key marking the correction messages the hook adds to the conversation. 

22_CORRECTION_META_KEY = "structured_output_correction"

24# A markdown code fence wrapping the whole reply, with an optional language tag such as `json`. 

25_CODE_FENCE_PATTERN = re.compile(r"^```[\w-]*[ \t]*\n(.*?)\n?[ \t]*```$", re.DOTALL)

28class StructuredOutputHook: 

29 """ 

30 Parse the Agent's final text reply as JSON and store it in a `State` key. 

31 

32 The hook runs at the `on_exit` hook point. When the Agent stops with a text reply (`exit_reason` is `"text"`), 

33 the hook parses the reply as JSON, validates it against `schema`, and stores the result under `state_key`. 

34 A markdown code fence around the whole reply (for example ```` ```json ````) is removed before parsing. 

35 With a Pydantic model as `schema`, the stored value is the validated model dumped to a dict. 

36 

37 When the reply is not valid JSON or does not match the schema, and retries are left, the hook adds a user 

38 message describing the error and keeps the Agent running so the model can correct its reply. Retries count 

39 toward `max_agent_steps`. When no retries are left, or the Agent stops on an exit condition other than a text 

40 reply, `None` is stored under `state_key`. 

41 

42 When an earlier `on_exit` hook has already set `continue_run`, the hook skips parsing and parses the reply at 

43 the next exit instead. 

44 

45 `on_exit` hooks don't run when the Agent stops because `max_agent_steps` is reached or a hook sets `stop_run`. 

46 If the hook never ran during the run, `state_key` is missing from the Agent's result, so read it with 

47 `result.get("structured_output")`. 

48 

49 `state_key` must be declared in the Agent's `state_schema`, which also makes it an output of the Agent. 

50 

51 The hook works with any chat generator. When the chat generator supports native structured output (for example 

52 `text_format` on `OpenAIResponsesChatGenerator`), pass the same schema there as well so the model is constrained 

53 to valid JSON and retries are rarely needed. 

54

84 def __init__( 

85 self, *, schema: type[BaseModel] | dict[str, Any], state_key: str = "structured_output", max_retries: int = 2 

86 ) -> None: 

87 """ 

88 Create a structured output hook. 

89 

90 :param schema: A Pydantic model or a JSON schema the final reply is validated against. 

91 :param state_key: The `State` key the parsed reply is stored under. It must be declared in the Agent's 

92 `state_schema`. Defaults to `"structured_output"`. 

93 :param max_retries: How many times the model is asked to correct a reply that is not valid JSON or does not 

94 match the schema. Defaults to 2. Set it to 0 to store `None` on the first failure. 

95 :raises ValueError: If `schema` is not a valid JSON schema, or if `max_retries` is negative. 

96 :raises TypeError: If `schema` is neither a Pydantic model nor a dict. 

97 """ 

98 if isinstance(schema, dict): 

99 try: 

100 Draft202012Validator.check_schema(schema) 

101 except SchemaError as e: 

102 raise ValueError(f"`schema` is not a valid JSON schema: {e.message}") from e 

103 self._validator: Draft202012Validator | None = Draft202012Validator(schema) 

104 self._model: type[BaseModel] | None = None 

105 json_schema = schema 

106 elif isinstance(schema, type) and issubclass(schema, BaseModel): 

107 self._validator = None 

108 self._model = schema 

109 json_schema = schema.model_json_schema() 

110 else: 

111 raise TypeError(f"`schema` must be a Pydantic model or a JSON schema dict, got {type(schema).__name__}.") 

112 if max_retries < 0: 

113 raise ValueError(f"`max_retries` must be 0 or greater, got {max_retries}.") 

114 

115 self.schema = schema 

116 self.state_key = state_key 

117 self.max_retries = max_retries 

118 self._schema_text = json.dumps(json_schema, indent=2) 

119 

120 def run(self, state: State) -> None: 

121 """ 

122 Parse the final reply and store it, or ask the model to correct it. 

123 

124 :param state: Agent state holding the conversation and the `exit_reason`. 

125 :raises ValueError: If `state_key` is not declared in the Agent's `state_schema`. 

126 """ 

127 if self.state_key not in state.schema: 

128 raise ValueError( 

129 f"StructuredOutputHook stores its result under '{self.state_key}', which is not in the Agent's " 

130 f"`state_schema`. Add it, for example `state_schema={{'{self.state_key}': {{'type': dict}}}}`." 

131 ) 

132 

133 # Another on_exit hook keeps the Agent running, so parse the reply at the next exit 

134 if state.data.get("continue_run"): 

135 return 

136 

137 exit_reason = state.data.get("exit_reason") 

138 if exit_reason != "text": 

139 logger.warning( 

140 "The Agent stopped with exit reason '{exit_reason}' instead of a text reply, so there is no " 

141 "structured output to parse. Storing None under '{state_key}'.", 

142 exit_reason=exit_reason, 

143 state_key=self.state_key, 

144 ) 

145 state.set(self.state_key, None) 

146 return 

147 

148 messages = state.data["messages"] 

149 parsed, error = self._parse(text=messages[-1].text) 

150 if error is None: 

151 state.set(self.state_key, parsed) 

152 return 

153 

154 state.set(self.state_key, None) 

155 retries_used = _count_corrections(messages=messages) 

156 if retries_used >= self.max_retries: 

157 logger.warning( 

158 "The Agent's final reply could not be parsed as structured output after {retries} retries: {error}", 

159 retries=retries_used, 

160 error=error, 

161 ) 

162 return 

163 

164 # Ask the model to correct its reply and keep the Agent running 

165 correction = ChatMessage.from_user( 

166 f"Your previous reply could not be used: {error}\n\n" 

167 f"Reply again with only JSON that matches this JSON schema:\n{self._schema_text}", 

168 meta={_CORRECTION_META_KEY: True}, 

169 ) 

170 state.set("messages", [correction]) 

171 state.set("continue_run", True)

173 def _parse(self, text: str | None) -> tuple[Any, str | None]: 

174 """ 

175 Parse and validate a reply. 

176 

177 :returns: The parsed value and `None` on success, or `None` and a description of the error on failure. 

178 """ 

179 if not text: 

180 return None, "The reply has no text." 

181 try: 

182 data = json.loads(_strip_code_fence(text=text)) 

183 except json.JSONDecodeError as e: 

184 return None, f"The reply is not valid JSON: {e}" 

185 

186 # self._model is not None when the schema is a Pydantic model 

187 if self._model is not None: 

188 try: 

189 model = self._model.model_validate(data) 

190 except ValidationError as e: 

191 return None, f"The reply does not match the schema: {e}" 

192 return model.model_dump(mode="json"), None 

193 

194 # self._validator is not None when the schema is a JSON schema dict 

195 if self._validator is not None and (validation_error := best_match(self._validator.iter_errors(data))): 

196 return None, f"The reply does not match the schema: {validation_error.message}" 

197 return data, None

199 def to_dict(self) -> dict[str, Any]: 

200 """ 

201 Serialize this hook to a dictionary. 

202 

203 :returns: Serialized representation of the hook. 

204 """ 

205 schema = self.schema if isinstance(self.schema, dict) else serialize_type(self.schema) 

206 return default_to_dict(self, schema=schema, state_key=self.state_key, max_retries=self.max_retries)

208 @classmethod 

209 def from_dict(cls, data: dict[str, Any]) -> "StructuredOutputHook": 

210 """ 

211 Create a hook from its serialized representation. 

212 

213 :param data: Serialized hook data. 

214 :returns: The deserialized hook. 

215 """ 

216 init_params = data.get("init_parameters", {}) 

217 if isinstance(init_params.get("schema"), str): 

218 data = {**data, "init_parameters": {**init_params, "schema": deserialize_type(init_params["schema"])}} 

219 return default_from_dict(cls, data=data)

222def _count_corrections(messages: list[ChatMessage]) -> int: 

223 """ 

224 Count the correction messages the hook added since the last message from the user. 

225 

226 Only the current turn is counted, so corrections from earlier turns of a multi-turn conversation don't use up 

227 the retries of the current one. 

228 """ 

229 count = 0 

230 # Walk back from the newest message 

231 for message in reversed(messages): 

232 # Skip all non-user messages, including the assistant's reply and any tool results 

233 if not message.is_from(ChatRole.USER): 

234 continue 

235 # The first untagged user message is the start of the current turn, so stop counting 

236 if not message.meta.get(_CORRECTION_META_KEY): 

237 break 

238 count += 1 

239 return count

242def _strip_code_fence(text: str) -> str: 

243 """ 

244 Remove a markdown code fence that wraps the whole reply. 

245 

246 Text with anything before or after the fence is returned unchanged. 

247 """ 

248 match = _CODE_FENCE_PATTERN.match(text.strip()) 

249 return match.group(1) if match else text
