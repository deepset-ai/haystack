5from typing import Any 

6 

7from haystack.components.generators.chat.types import ChatGenerator 

8 

9# The provider-neutral generation parameters that Haystack components can request from Chat Generators. 

10# The chosen name is based on OpenAI's Responses API. 

11_HAYSTACK_GENERATION_PARAMETERS = frozenset({"max_output_tokens"}) 

12 

13 

14def _convert_haystack_generation_kwargs( 

15 chat_generator: ChatGenerator, haystack_generation_kwargs: dict[str, Any] 

16) -> dict[str, Any]: 

17 """ 

18 Convert provider-neutral Haystack generation parameters for a Chat Generator. 

19 

20 Chat Generators advertise supported parameters through a private class-level mapping from the canonical Haystack 

21 name to the provider-specific name. Parameters not advertised by the generator are omitted, allowing callers to 

22 provide a fallback for generators that do not expose this optional capability. 

23 

24 :param chat_generator: The Chat Generator that will receive the converted parameters. 

25 :param haystack_generation_kwargs: Generation parameters using Haystack's canonical names. 

26 :returns: The supported parameters converted to their provider-specific names. 

27 :raises ValueError: If a parameter is not part of Haystack's canonical vocabulary. 

28 """ 

29 unknown_parameters = haystack_generation_kwargs.keys() - _HAYSTACK_GENERATION_PARAMETERS 

30 if unknown_parameters: 

31 unknown = ", ".join(sorted(unknown_parameters)) 

32 msg = f"Unknown Haystack generation parameter(s): {unknown}" 

33 raise ValueError(msg) 

34 

35 parameter_mapping = getattr(chat_generator, "_HAYSTACK_TO_PROVIDER_GENERATION_KWARGS", {}) 

36 

37 return { 

38 provider_name: haystack_generation_kwargs[haystack_name] 

39 for haystack_name, provider_name in parameter_mapping.items() 

40 if haystack_name in haystack_generation_kwargs 

41 }
